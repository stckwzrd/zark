Асинхронность

std/async — однопоточный исполнитель поверх поллера рантайма (epoll в Linux, kqueue в macOS). Это обычный Zark плюс несколько вызовов vrt_poller_*; в компиляторе нет ничего async-специфичного, кроме генерации фрейма async-функции.

#Фьючер

Фьючер — любой struct с двумя методами:

Poll<T> poll(Context* cx);
void cancel();

poll возвращает Ready(value) или Pending. Вернуть Pending — значит обещать разбудить задачу: либо вызвать cx.waker.wake(), либо взвести Readable, Writable или Sleep. Фьючер, который не делает ни того, ни другого, останавливает исполнитель. cancel() освобождает то, что взвёл последний poll; spawn и block_on требуют его, а async-функция получает его от компилятора.

import "std/async" as async;

struct Countdown {
    i64 n;
    void cancel() {}
    Poll<i64> poll(Context* cx) {
        if (this->n == 0) { return .Ready(42); }
        this->n -= 1;
        cx.waker.wake();
        return .Pending;
    }
}

!void main() {
    auto ex = async::Executor::init(c_allocator())?;
    defer ex.deinit();
    auto f = Countdown{ .n = 3 };
    println(ex.block_on<Countdown, i64>(&f)?);
}
Вывод
42

#Executor, spawn, join, select

Executor::init(a) владеет поллером и задачами. ex.block_on(&f) крутит фьючер и запущенные им задачи, пока фьючер не станет готов; если прогресса нет, возвращается error.Deadlock. ex.spawn(f) забирает фьючер по move, кладёт его в память исполнителя и возвращает JoinHandle<T>. Тип результата spawn и block_on выводится из poll фьючера.

import "std/async" as async;

struct Ready {
    i64 v;
    void cancel() {}
    Poll<i64> poll(Context* cx) { return .Ready(this->v); }
}

struct Never {
    void cancel() {}
    Poll<i64> poll(Context* cx) { return .Pending; }
}

!void main() {
    auto ex = async::Executor::init(c_allocator())?;
    defer ex.deinit();
    auto a = ex.spawn(Ready{ .v = 20 })?;
    auto b = ex.spawn(Ready{ .v = 22 })?;
    auto j = async::join(a, b);
    const auto p = ex.block_on(&j)?;
    println(p.first + p.second);

    auto s = async::select(ex.spawn(Ready{ .v = 7 })?, ex.spawn(Never{})?);
    const auto r = ex.block_on(&s)?;
    match (r) {
        .First(v) => println(v),
        .Second(_) => println(-1),
    }
}
Вывод
42
7

join(a, b) даёт Pair<A, B>, когда готовы оба handle. select(a, b) даёт Select2Result<A, B> от того, кто завершился первым, и отменяет второго. JoinHandle одноразовый: take() забирает результат, detach() оставляет задачу работать без результата, и после любого из них handle невалиден. Копия использованного handle паникует.

#Async-функции и отмена

async-функция выполняет тело до await и возвращает Pending-фрейм. await f опрашивает f и приостанавливает фрейм. Компилятор превращает тело в struct-фрейм и генерирует poll и cancel:

import "std/async" as async;

struct Never {
    void cancel() {}
    Poll<bool> poll(Context* cx) { return .Pending; }
}

i64 cleaned = 0;

async !void worker() {
    defer cleaned += 1;
    auto n = Never{};
    await n;
}

!void main() {
    {
        auto ex = async::Executor::init(c_allocator())?;
        defer ex.deinit();
        auto h = ex.spawn(worker())?;
        ex.run_batch();
        println(cleaned);
        h.cancel();
        println(cleaned);
    }
    println(cleaned);
}
Вывод
0
1
1

Отмена — это незавершённая операция, а не обычный возврат: сгенерированный cancel() выполняет и defer, и errdefer всех активных на точке приостановки областей, изнутри наружу, отменяет фьючер, которого ждали, и помечает фрейм завершённым. Повторный вызов идемпотентен и ничего не делает. Executor.deinit() отменяет каждую живую задачу перед освобождением, и каждая задача отменяется ровно один раз. detach() не отменяет: задача продолжает работать.

cancel() у JoinHandle — другое: он инвалидирует handle и выбрасывает незабранный результат, поэтому второй вызов паникует, как и take(). select отменяет проигравший handle; join отменяет обе стороны, кроме той, что уже отдала значение и была забрана.

В std/async также есть TcpListener, TcpStream, Readable, Writable и Sleep. Исполнитель однопоточный: он опрашивает задачи по одной; каналов и многопоточного исполнителя пока нет.
